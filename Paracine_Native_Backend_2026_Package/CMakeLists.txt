cmake_minimum_required(VERSION 3.20)
project(ParacineNativeBackend VERSION 1.0 LANGUAGES CXX)
add_executable(ParacineCompiler ParacineCompiler_2026_Native_Backend_Optimized.cpp)
set_target_properties(ParacineCompiler PROPERTIES CXX_STANDARD 23 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF OUTPUT_NAME paracine)
if(MSVC)
    target_compile_options(ParacineCompiler PRIVATE /EHsc /utf-8 /W3 /bigobj)
else()
    target_compile_options(ParacineCompiler PRIVATE -Wall -Wno-unused-function)
endif()
enable_testing()
add_test(NAME compiler_selftest COMMAND ParacineCompiler selftest)
add_test(NAME differential_arithmetic_and_branches COMMAND ParacineCompiler test ${CMAKE_CURRENT_SOURCE_DIR}/tests/arithmetic_and_branches.pcn)
add_test(NAME differential_memory_and_calls COMMAND ParacineCompiler test ${CMAKE_CURRENT_SOURCE_DIR}/tests/memory_and_calls.pcn)
